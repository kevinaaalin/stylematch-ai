# 05 Runtime SQL

```sql
CREATE TABLE workflow_runtime(

runtime_id BIGINT PRIMARY KEY,

workflow_id BIGINT NOT NULL,

case_id BIGINT NOT NULL,

current_state BIGINT,

runtime_status VARCHAR(30),

started_at TIMESTAMP,

updated_at TIMESTAMP

);
```

---

```sql
CREATE TABLE workflow_stage(

stage_id BIGINT PRIMARY KEY,

workflow_id BIGINT,

stage_code VARCHAR(40),

stage_name VARCHAR(100),

sequence_no INT,

gate_id BIGINT

);
```

---

```sql
CREATE TABLE workflow_decision(

decision_id BIGINT PRIMARY KEY,

runtime_id BIGINT,

decision_type VARCHAR(40),

decision_result VARCHAR(20),

operator_id BIGINT,

decision_time TIMESTAMP

);
```
