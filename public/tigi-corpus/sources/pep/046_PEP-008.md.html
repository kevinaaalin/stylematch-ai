# 05 Integration SQL

```sql
CREATE TABLE integration_connector(

connector_id BIGINT PRIMARY KEY,

connector_code VARCHAR(50),

connector_name VARCHAR(200),

connector_type VARCHAR(50),

status VARCHAR(30),

created_at TIMESTAMP

);
```

```sql
CREATE TABLE integration_endpoint(

endpoint_id BIGINT PRIMARY KEY,

connector_id BIGINT,

endpoint_url VARCHAR(500),

http_method VARCHAR(10),

authentication_type VARCHAR(30),

status VARCHAR(30)

);
```

```sql
CREATE TABLE integration_webhook(

webhook_id BIGINT PRIMARY KEY,

connector_id BIGINT,

event_name VARCHAR(100),

callback_url VARCHAR(500),

secret_key VARCHAR(255),

status VARCHAR(30)

);
```
